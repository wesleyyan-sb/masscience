export function planMinicutEnergyTarget(params) {
  const { startingBf, currentBf, weightKg, tdee } = params;
  
  // Fat mass to lose strictly to baseline BF
  const fatMassDeltaKg = calcFatLossToBaseline(currentBf, startingBf, weightKg);
  const totalDeficitNeeded = fatMassDeltaKg * 7700;
  
  // Preferred 21-day horizon with STRICT 650 kcal cap
  let dailyDeficit = Math.round(totalDeficitNeeded / 21);
  dailyDeficit = Math.min(dailyDeficit, 650); // NON-NEGOTIABLE
  
  return {
    recommendedCalories: tdee - dailyDeficit,
    deficitKcal: dailyDeficit,
    projectedDays: Math.ceil(totalDeficitNeeded / dailyDeficit)
  };
}
